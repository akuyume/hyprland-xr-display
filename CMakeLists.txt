cmake_minimum_required(VERSION 3.20)
project(xr_hyprland VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WL REQUIRED IMPORTED_TARGET wayland-client wayland-egl egl glesv2 gbm libdrm)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(PROTOCOLS wlr-screencopy-unstable-v1 wlr-layer-shell-unstable-v1 linux-dmabuf-v1 xdg-shell
  ext-foreign-toplevel-list-v1 ext-image-capture-source-v1 ext-image-copy-capture-v1
  hyprland-toplevel-mapping-v1 wlr-foreign-toplevel-management-unstable-v1)
foreach(P IN LISTS PROTOCOLS)
  add_custom_command(OUTPUT ${P}-client.h ${P}-protocol.c
    COMMAND ${WAYLAND_SCANNER} client-header ${CMAKE_CURRENT_SOURCE_DIR}/protocols/${P}.xml ${P}-client.h
    COMMAND ${WAYLAND_SCANNER} private-code ${CMAKE_CURRENT_SOURCE_DIR}/protocols/${P}.xml ${P}-protocol.c
    DEPENDS protocols/${P}.xml)
  list(APPEND GENERATED ${P}-client.h ${P}-protocol.c)
endforeach()
add_library(xr-core src/pose.cpp src/config.cpp src/hyprland.cpp src/stereo.cpp)
target_include_directories(xr-core PUBLIC src)
target_compile_options(xr-core PRIVATE -Wall -Wextra -Wpedantic)
add_library(xr-protocols STATIC ${GENERATED})
target_include_directories(xr-protocols PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(xr-protocols PUBLIC PkgConfig::WL)
add_library(xr-windows src/windows.cpp src/scene.cpp)
target_link_libraries(xr-windows PUBLIC xr-core xr-protocols)
target_compile_options(xr-windows PRIVATE -Wall -Wextra -Wpedantic)
add_executable(xr-hyprland src/main.cpp src/wayland.cpp)
target_link_libraries(xr-hyprland PRIVATE xr-windows)
target_compile_options(xr-hyprland PRIVATE -Wall -Wextra -Wpedantic)
add_executable(xr-hyprlandctl src/ctl.cpp)
target_link_libraries(xr-hyprlandctl PRIVATE xr-windows)
add_executable(xr-pose src/pose_main.cpp)
target_link_libraries(xr-pose PRIVATE xr-core)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME monitor-power COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/monitor_power.py)
add_test(NAME desktop-transfer COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/desktop_transfer.py)
add_executable(xr-tests tests/core.cpp)
target_link_libraries(xr-tests PRIVATE xr-core PkgConfig::WL)
add_test(NAME core COMMAND xr-tests)
add_executable(xr-render-tests tests/render.cpp)
target_link_libraries(xr-render-tests PRIVATE xr-core PkgConfig::WL)
add_test(NAME render COMMAND xr-render-tests)
set_tests_properties(render PROPERTIES SKIP_RETURN_CODE 77)
install(TARGETS xr-hyprland xr-hyprlandctl xr-pose RUNTIME DESTINATION bin)
install(PROGRAMS scripts/xr-display DESTINATION bin)
install(FILES config.ini DESTINATION share/xr-hyprland)
install(FILES scripts/xr-hyprland.service DESTINATION share/xr-hyprland)
